// src/core/lib/foodSovereigntyIndex.ts — NUEVO MÓDULO P0

interface FoodSovereigntyIndex {
  communityId: string;
  timestamp: number;
  period: 'epoch' | 'seasonal' | 'annual';
  
  // 10 Componentes (0-100 cada uno)
  components: {
    // Autonomía Nutricional (4)
    autonomyCaloric: number;         // % calorías locales vs consumo total
    autonomyProtein: number;         // % proteína local vs consumo total
    autonomyMicronutrient: number;   // % micronutrientes locales vs RDI
    dietaryDiversity: number;        // Shannon index normalizado 0-100
    
    // Circularidad (2)
    cycleClosure: number;            // % residuos orgánicos → insumos producción
    entropicEfficiency: number;      // ξ normalizado (0-100)
    
    // Liberación (2)
    laborReduction: number;          // Horas trabajo lábil per cápita vs baseline (0-100)
    monetaryReduction: number;       // USD/kg alimento vs baseline (0-100)
    
    // Gobernanza (2)
    governanceQuality: number;       // CDS maturity + evidence VERIFIED% + Symbiosky health
    duplicability: number;           // Kit 72H deploy success rate + handshake completion
  };
  
  // Composite FSI (media geométrica 10 componentes)
  FSI: number; // 0-100
  
  // Targets por Fase HSCSG
  targets: {
    PhaseA_ProtoCO: 30;    // 0-3 meses: baseline + diseño
    PhaseA_Emergente: 45;  // 3-9 meses: producción piloto
    PhaseB_Estructurado: 65; // 9-18 meses: sistema operativo
    PhaseC_Maduro: 85;     // 18-36 meses: resiliente + duplicable
    PhaseD_Federado: 95;   // Continuo: federación ontogenética
  };
  
  // Tendencia
  trend: 'improving' | 'stable' | 'declining';
  velocity: number; // puntos FSI por epoch (28d)
  
  // Evidencia para gobernanza
  evidence: EvidenceRecord[]; // Fact Bands VERIFIED_AUTONOMY, etc.
}

// Cálculo FSI
function calculateFSI(components: FSIComponents): number {
  const values = Object.values(components);
  // Media geométrica: penaliza componentes en 0
  const product = values.reduce((a, b) => a * (b || 0.01), 1);
  return Math.round(Math.pow(product, 1/values.length));
}

// Integración con Evidence Model
function emitFSIEvidence(fsi: FoodSovereigntyIndex): EvidenceRecord {
  return {
    type: 'fsi_measurement',
    band: fsi.FSI >= 65 ? MatrioskaFactBand.VERIFIED_AUTONOMY : 
          fsi.FSI >= 40 ? MatrioskaFactBand.PROBABLE_AUTONOMY :
          MatrioskaFactBand.EXTERNALIZED_COST,
    score: fsi.FSI / 100,
    detail: `FSI:${fsi.FSI} components:${JSON.stringify(fsi.components)}`,
    timestamp: fsi.timestamp,
    communityId: fsi.communityId
  };
}
